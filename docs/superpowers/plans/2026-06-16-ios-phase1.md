import Foundation
import Security

final class KeychainManager {
    static let shared = KeychainManager()

    func save(key: String, value: String) {
        guard let data = value.data(using: .utf8) else { return }
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrAccount as String: key,
            kSecValueData as String: data
        ]
        SecItemDelete(query as CFDictionary)
        SecItemAdd(query as CFDictionary, nil)
    }

    func read(key: String) -> String? {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrAccount as String: key,
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne
        ]
        var result: AnyObject?
        guard SecItemCopyMatching(query as CFDictionary, &result) == errSecSuccess,
              let data = result as? Data else { return nil }
        return String(data: data, encoding: .utf8)
    }

    func delete(key: String) {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrAccount as String: key
        ]
        SecItemDelete(query as CFDictionary)
    }
}

import Foundation

final class APIClient {
    static let shared = APIClient()
    // 从 Info.plist 读取，默认 production；开发时可设 http://localhost:8080
    var baseURL: String = Bundle.main.object(forInfoDictionaryKey: "KN_API_BASE_URL") as? String ?? "https://api.knshark.com"

    private var accessToken: String? {
        KeychainManager.shared.read(key: "access_token")
    }

    func post<T: Decodable>(_ path: String, body: [String: Any]) async throws -> T {
        var request = URLRequest(url: URL(string: "\(baseURL)\(path)")!)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        if let token = accessToken {
            request.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")
        }
        request.httpBody = try JSONSerialization.data(withJSONObject: body)

        let (data, _) = try await URLSession.shared.data(for: request)
        return try JSONDecoder().decode(T.self, from: data)
    }

    func get<T: Decodable>(_ path: String) async throws -> T {
        var request = URLRequest(url: URL(string: "\(baseURL)\(path)")!)
        if let token = accessToken {
            request.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")
        }
        let (data, _) = try await URLSession.shared.data(for: request)
        return try JSONDecoder().decode(T.self, from: data)
    }
}

```swift
import Foundation

@MainActor
final class WebSocketClient: ObservableObject {
    private var task: URLSessionWebSocketTask?
    private var session: URLSession
    private var pingTimer: Timer?
    private var reconnectAttempts = 0
    private let maxReconnectDelay: TimeInterval = 30

    @Published var isConnected = false
    @Published var messages: [ServerMessage] = []
    var onMessage: ((ServerMessage) -> Void)?

    init() {
        let config = URLSessionConfiguration.default
        self.session = URLSession(configuration: config)
    }

    func connect(token: String? = nil, deviceToken: String? = nil, code: String? = nil) {
        // 保存参数用于重连
        self.lastToken = token
        self.lastDeviceToken = deviceToken
        self.lastCode = code
        connectInternal()
    }

    private var lastToken: String?
    private var lastDeviceToken: String?
    private var lastCode: String?
    private var lastAckSeq: UInt64 = 0  // 用于重连时上报

    private func connectInternal() {
        // WSS URL 从 APIClient baseURL 派生
        let baseURL = APIClient.shared.baseURL
            .replacingOccurrences(of: "https://", with: "wss://")
            .replacingOccurrences(of: "http://", with: "ws://")
        // 云服务 WebSocket 注册在 /ws（非 /v1/ws），与 REST API /api/v1/* 分离
        let url = URL(string: baseURL.replacingOccurrences(of: "/api/v1", with: "") + "/ws")!
        var request = URLRequest(url: url)
        // 通过 HTTP Header 鉴权（不在 URL query string 中）
        if let t = lastToken {
            request.setValue("Bearer \(t)", forHTTPHeaderField: "Authorization")
            request.setValue("ios", forHTTPHeaderField: "X-KN-Role")
        }
        if let d = lastDeviceToken {
            request.setValue("Bearer \(d)", forHTTPHeaderField: "Authorization")
        }
        if let c = lastCode {
            request.setValue("Bearer \(c)", forHTTPHeaderField: "Authorization")
        }
        request.setValue("1", forHTTPHeaderField: "X-KN-Protocol-Version")

        task = session.webSocketTask(with: request)
        task?.resume()
        startPing()
        receive()

        // 重连后上报 last_ack_seq，云端据此补推离线消息
        if lastAckSeq > 0 {
            send(.ack(msgSeq: lastAckSeq))
        }
    }

    func disconnect() {
        pingTimer?.invalidate()
        task?.cancel(with: .goingAway, reason: nil)
        isConnected = false
    }

    func send(_ message: ClientMessage) {
        guard let data = try? JSONEncoder().encode(message),
              let json = String(data: data, encoding: .utf8) else { return }
        task?.send(.string(json)) { _ in }
    }

    private func receive() {
        task?.receive { [weak self] result in
            guard let self else { return }
            switch result {
            case .success(let message):
                switch message {
                case .string(let text):
                    if let msg = try? JSONDecoder().decode(ServerMessage.self, from: text.data(using: .utf8)!) {
                        self.messages.append(msg)
                        self.onMessage?(msg)
                    }
                default: break
                }
                self.receive()
            case .failure:
                self.isConnected = false
                self.scheduleReconnect()
            }
        }
    }

    private func startPing() {
        // 两层心跳：① WebSocket 协议 ping (keepalive) ② 应用层 JSON ping
        pingTimer = Timer.scheduledTimer(withTimeInterval: 15, repeats: true) { [weak self] _ in
            self?.task?.sendPing { _ in }  // 传输层
            self?.send(.ping)              // 应用层 {type: "ping"}
        }
    }

    private func scheduleReconnect() {
        let delay = min(pow(2.0, Double(reconnectAttempts)), maxReconnectDelay)
        reconnectAttempts += 1
        Task {
            try? await Task.sleep(nanoseconds: UInt64(delay * 1_000_000_000))
            connectInternal()  // 用保存的参数重连
        }
    }
}

// MARK: - Message Types
enum ClientMessage: Codable {
    // Will be replaced with proper types once connected
    case ping, startSession(sessionId: String, tool: String, profile: String?, cwd: String), input(sessionId: String, text: String)
}

// session_id 由发起方本地生成，无需云端协调
// 用法: let sid = "s_" + NanoID.generate(12) → startSession(sessionId: sid, tool: "claude", ...)
// 依赖: NanoID 库 (SPM: https://github.com/nimble-dev/nanoid-swift)，或手写 10 行 base62 实现

enum ServerMessage: Codable {
    // Will be expanded
    case connected(wsSessionId: String), output(sessionId: String, ansiText: String), errorNotify(code: String, message: String)
}
```

import SwiftUI

@MainActor
final class AuthViewModel: ObservableObject {
    @Published var email = ""
    @Published var password = ""
    @Published var isLoading = false
    @Published var errorMessage: String?

    func register() async -> Bool {
        isLoading = true; errorMessage = nil
        defer { isLoading = false }
        do {
            // 服务器返回: {"code":0,"message":"ok","data":{"access_token":"...","refresh_token":"...","user_id":1}}
            let resp: ApiResponse<AuthResult> = try await APIClient.shared.post(
                "/api/v1/auth/register", body: ["email": email, "password": password])
            if resp.code != 0 { errorMessage = resp.message; return false }
            if let access = resp.data?.accessToken {
                KeychainManager.shared.save(key: "access_token", value: access)
                KeychainManager.shared.save(key: "refresh_token", value: resp.data!.refreshToken)
            }
            return true
        } catch { errorMessage = error.localizedDescription; return false }
    }

    func login() async -> Bool {
        isLoading = true; errorMessage = nil
        defer { isLoading = false }
        do {
            let resp: ApiResponse<AuthResult> = try await APIClient.shared.post(
                "/api/v1/auth/login", body: ["email": email, "password": password])
            if resp.code != 0 { errorMessage = resp.message; return false }
            if let access = resp.data?.accessToken {
                KeychainManager.shared.save(key: "access_token", value: access)
                KeychainManager.shared.save(key: "refresh_token", value: resp.data!.refreshToken)
            }
            return true
        } catch { errorMessage = error.localizedDescription; return false }
    }
}

import SwiftUI
import WebKit

struct TerminalView: UIViewRepresentable {
    @ObservedObject var viewModel: TerminalViewModel

    func makeUIView(context: Context) -> WKWebView {
        let config = WKWebViewConfiguration()
        config.userContentController.add(context.coordinator, name: "terminalInput")
        config.userContentController.add(context.coordinator, name: "terminalResize")

        let webView = WKWebView(frame: .zero, configuration: config)
        webView.isOpaque = false
        webView.backgroundColor = UIColor(red: 0.04, green: 0.05, blue: 0.06, alpha: 1)
        webView.scrollView.isScrollEnabled = false

        // Load bundled terminal.html
        if let url = Bundle.main.url(forResource: "terminal", withExtension: "html") {
            webView.loadFileURL(url, allowingReadAccessTo: url.deletingLastPathComponent())
        }
        return webView
    }

    func updateUIView(_ webView: WKWebView, context: Context) {
        if let ansi = viewModel.pendingANSI, !ansi.isEmpty {
            // base64 编码避免 ANSI 转义序列破坏 JavaScript 字符串
            let base64 = ansi.data(using: .utf8)!.base64EncodedString()
            webView.evaluateJavaScript("window.writeANSIBase64('\(base64)');")
            viewModel.pendingANSI = nil
        }
    }

    func makeCoordinator() -> Coordinator { Coordinator(viewModel: viewModel) }

    class Coordinator: NSObject, WKScriptMessageHandler {
        let viewModel: TerminalViewModel
        init(viewModel: TerminalViewModel) { self.viewModel = viewModel }

        func userContentController(_ uc: WKUserContentController, didReceive message: WKScriptMessage) {
            if message.name == "terminalInput", let text = message.body as? String {
                viewModel.sendInput(text)
            }
            if message.name == "terminalResize",
               let body = message.body as? [String: Int],
               let cols = body["cols"], let rows = body["rows"] {
                viewModel.terminalSize = (cols: cols, rows: rows)
            }
        }
    }
}

@MainActor
final class TerminalViewModel: ObservableObject {
    @Published var pendingANSI: String?
    @Published var inputText = ""
    @Published var isDirectMode = false  // [⚡] 直通模式
    @Published var sessionId: String?    // 当前活跃 session_id

    func startSession(tool: String, profile: String?, cwd: String) {
        let sid = "s_" + NanoID.generate(12)  // 去中心化 session_id
        sessionId = sid
        ws?.send(.startSession(sessionId: sid, tool: tool, profile: profile, cwd: cwd))
    }

    func sendInput(_ text: String) {
        ws?.send(.input(sessionId: currentSessionId ?? "", text: text))
    }

    func sendManual() {
        guard !inputText.isEmpty else { return }
        sendInput(inputText)
        inputText = ""
    }

    var currentSessionId: String?
}

```swift
@MainActor
final class AppState: ObservableObject {
    @Published var isLoggedIn: Bool
    @Published var terminalVM = TerminalViewModel()
    let ws = WebSocketClient()

    init() {
        let token = KeychainManager.shared.read(key: "access_token")
        isLoggedIn = token != nil && !isTokenExpired(token!)
        if isLoggedIn {
            ws.connect(token: token!)
            terminalVM.ws = ws
        }
    }

    /// JWT 过期检测：解析 payload 中的 exp 字段比较当前时间
    private func isTokenExpired(_ token: String) -> Bool {
        let parts = token.split(separator: ".")
        guard parts.count == 3,
              let payloadData = Data(base64Encoded: String(parts[1]).padding(
                  toLength: ((String(parts[1]).count + 3) / 4) * 4,
                  withPad: "=", startingAt: 0)),
              let json = try? JSONSerialization.jsonObject(with: payloadData) as? [String: Any],
              let exp = json["exp"] as? TimeInterval else {
            return true // 解析失败 → 视为过期
        }
        return Date().timeIntervalSince1970 > exp
    }

    func didLogin(accessToken: String) {
        KeychainManager.shared.save(key: "access_token", value: accessToken)
        isLoggedIn = true
        ws.connect(token: accessToken)
        terminalVM.ws = ws
    }

    func logout() {
        ws.disconnect()
        KeychainManager.shared.delete(key: "access_token")
        isLoggedIn = false
    }
}
```

- [ ] **Step 3: 直通模式 — WKWebView 处理 + CJK IME 保护**

在 `terminal.html` 中，直通模式下 `term.onData()` 直接回调 Swift。CJK IME（中文/日文/韩文输入法）在 composing 期间会频繁触发 `onData`，需通过 `compositionstart`/`compositionend` 事件做缓冲保护：

```js
// terminal.html 中添加:
var directMode = false;
var imeComposing = false;    // IME 激活中 → 缓冲输入，不立即发送
var imeBuffer = '';

// 监听 IME 事件（WKWebView 支持 compositionstart/end）
document.addEventListener('compositionstart', function() {
    imeComposing = true;
    imeBuffer = '';
});
document.addEventListener('compositionend', function(e) {
    imeComposing = false;
    // 只发送 IME 最终确认的字符
    if (directMode && imeBuffer) {
        window.webkit.messageHandlers.terminalInput.postMessage(imeBuffer);
        imeBuffer = '';
    }
});

window.setDirectMode = function(enabled) {
    directMode = enabled;
    imeComposing = false;
    imeBuffer = '';
};

term.onData(function(data) {
    if (directMode) {
        if (imeComposing) {
            imeBuffer += data;   // composing 期间缓冲，不发送
        } else {
            // 直通: 非 IME 键盘事件直接进入 PTY
            window.webkit.messageHandlers.terminalInput.postMessage(data);
        }
    }
    // 默认模式: 不处理键盘，由 Swift 输入栏控制
});
```
